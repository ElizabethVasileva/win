    FOREIGN KEY (category_id) REFERENCES product_categories(id) ON DELETE CASCADE ON UPDATE CASCADE
);

-- заказы
CREATE TABLE orders (
    id INT NOT NULL PRIMARY KEY,
    order_date TEXT,
    delivery_date TEXT,
    receipt_code INT,
    client_id INT,
    pickup_point_id INT,
    status_id INT,
    FOREIGN KEY (client_id) REFERENCES users(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (pickup_point_id) REFERENCES pickup_points(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (status_id) REFERENCES order_statuses(id) ON DELETE CASCADE ON UPDATE CASCADE
);

# 1) Проверка пользователя по логину и паролю
def check_user(login, password):
    try:
        # Подключаемся к БД и создаём курсор для SQL-запросов
        conn = get_connection()
        cursor = conn.cursor()

        # Поиск пользователя по логину и паролю, получаем роль и ФИО
        sql = """
        SELECT r.name, u.surname, u.name, u.patronymic
        FROM users u
        JOIN roles r ON u.role_id = r.id
        WHERE u.login = %s AND u.password = %s
        """
        cursor.execute(sql, (login, password)) # выполнение запроса с параметрами логин и пароль

        result = cursor.fetchone() # получение одной строки результата
        conn.close() # закрытие соединения с БД
        return result # возвращение результата(пользователя)

    # Ошибка подключения или запроса к БД
    except pymysql.MySQLError:
        return None

# 3) Универсальный запрос для поиска, фильтра и сортировки
def get_products_filtered(poisk_text="", postavshik="Все поставщики", order=None):
    conn = get_connection()
    cursor = conn.cursor()

    sql = """
    SELECT 
        p.id,
        pc.name,
        p.article,
        p.product_name,
        p.description,
        m.name,
        s.name,
        p.price,
        p.unit,
        p.stock_quantity,
        p.discount,
        p.photo
    FROM products p
    JOIN product_categories pc ON p.category_id = pc.id
    JOIN manufacturers m ON p.manufacturer_id = m.id
    JOIN suppliers s ON p.supplier_id = s.id
    WHERE 1=1
    """

    params = []

    # Поиск по всем доступным данным и нескольким атрибутам одновременно (админ и менеджер)
    if poisk_text:
        words = poisk_text.split() # разбиваем на слова
        for word in words:
            sql += """
            AND (
                pc.name LIKE %s OR
                p.article LIKE %s OR
                p.product_name LIKE %s OR
                p.description LIKE %s OR
                m.name LIKE %s OR
                s.name LIKE %s
            )
            """
            pattern = f"%{word}%"
            params.extend([pattern] * 6)

    # Фильтр по поставщику
    if postavshik and postavshik != "Все поставщики":
        sql += " AND s.name = %s"
        params.append(postavshik)

    # Сортировка товаров по кол-во на складе ("DESC" - по убыванию, "ASC" - по возрастанию)
    if order == "ASC":
        sql += " ORDER BY p.stock_quantity ASC"
    elif order == "DESC":
        sql += " ORDER BY p.stock_quantity DESC"

    cursor.execute(sql, params) # выполнение запроса
    result = cursor.fetchall() # получение всех строк результата
    conn.close() # закрытия соединения с БД
    return result # возращение результата запроса (список кортежей)

    SELECT 
        o.id,
        o.order_date,
        o.delivery_date,
        o.receipt_code,
        CONCAT(u.surname, ' ', u.name, ' ', IFNULL(u.patronymic, '')) AS client_fio,

        o.pickup_point_id,
        o.client_id,
        o.status_id
    FROM orders o

    JOIN city c ON pp.city_id = c.id
    JOIN streets s ON pp.street_id = s.id

    ORDER BY o.id
    """

    result = cursor.fetchall()

        for proiz_id, name in get_proizvoditel():

    def save_product(self):
        article = self.articleEdit.text().strip()
        name = self.nameEdit.text().strip()
        description = self.opisanieEdit.text().strip()
        unit = self.izmerenieEdit.text().strip()

        price = self.priceSpinBox.value()
        quantity = self.kolvoSpinBox.value()
        discount = self.skidkaSpinBox.value()

        category_id = self.categoryComboBox.currentData()
        proiz_id = self.proizvoditelComboBox.currentData()
        post_id = self.postavshikComboBox.currentData()

        photo_path = self.current_photo if self.current_photo else "picture.png"

            QMessageBox.warning(self, "Ошибка ввода", "Введите артикул товара.")
            return

            QMessageBox.warning(self, "Ошибка ввода", "Введите наименование товара.")
            return

        if not unit:
            QMessageBox.warning(self, "Ошибка ввода", "Введите единицу измерения товара.")
            return

        if price <= 0:
            QMessageBox.warning(self, "Ошибка ввода", "Цена должна быть больше 0.")
            return

        if quantity < 0:
            QMessageBox.warning(self, "Ошибка ввода", "Количество не может быть отрицательным.")
            return

        if discount < 0 or discount > 100:
            QMessageBox.warning(self, "Ошибка ввода", "Скидка должна быть в диапазоне от 0 до 100%.")
            return

        try:
            if self.product_id: # если это режим РЕДАКТИРОВАНИЯ
                update_product(
                    self.product_id, article, name, category_id, description,
                    proiz_id, post_id, price, unit, quantity, discount, photo_path
                )

                )

            QMessageBox.critical(

                f"Не удалось сохранить товар.\nОшибка: {str(e)}"
            )

        # показываем фото
        if self.current_photo:
            image_path = os.path.join("images", self.current_photo)
            if os.path.exists(image_path):
                pixmap = QPixmap(image_path)
                pixmap = pixmap.scaled(
                    self.imageLabel.width(),
                    self.imageLabel.height(),
                    Qt.AspectRatioMode.KeepAspectRatio
                )
                self.imageLabel.setPixmap(pixmap)

    # 4) Выбор картинки (копирует картинку в папку images и сохраняет имя файла в БД)
    def choose_image(self):
        file_path, _ = QFileDialog.getOpenFileName(
            self,
            "Выбрать изображение",
            "",
            "Images (*.png *.jpg *.jpeg)"
        )

        if not file_path:
            return

        try:
            os.makedirs("images", exist_ok=True)
            file_name = os.path.basename(file_path)
            save_path = os.path.join("images", file_name)

            # УДАЛЯЕМ старую картинку, если она была
            if self.current_photo and self.current_photo != "picture.png":
                old_path = os.path.join("images", self.current_photo)
                # если выбрали ДРУГУЮ картинку
                if os.path.exists(old_path) and self.current_photo != file_name:
                    os.remove(old_path)

            # если файл не из папки images — копируем
            if os.path.abspath(file_path) != os.path.abspath(save_path):
                shutil.copy(file_path, save_path)

            # сохраняем имя файла
            self.current_photo = file_name

            pixmap = QPixmap(save_path)
            pixmap = pixmap.scaled(
                self.imageLabel.width(),
                self.imageLabel.height(),
                Qt.AspectRatioMode.KeepAspectRatio
            )
            self.imageLabel.setPixmap(pixmap)

        except Exception as e:
            QMessageBox.critical(
                self,
                "Ошибка изображения",
                f"Не удалось загрузить изображение.\nОшибка: {str(e)}"
            )

class OrderCard(QFrame):
    order_updated = pyqtSignal()

    def __init__(self, order_data, role, parent=None):
        super().__init__(parent)
        uic.loadUi("ui/orders.ui", self)

        self.order_id = order_data[0]
        self.role = role

        # Заполняем данные

        self.status_orderLabel.setText(f"Статус заказа: {order_data[6]}")
        self.punktLabel.setText(f"Адрес пункта выдачи: {order_data[4]}")
        self.date_deliveryLabel.setText(f"Дата заказа: {order_data[1]}")

        # Дата доставки
        if order_data[2] and order_data[2].strip():
            self.deliveryLabel.setText(f"Дата доставки\n{order_data[2]}")
        else:
            self.deliveryLabel.setText("Дата доставки\nне указана")

        # Для менеджера и администратора добавляем выбор статуса
        if role in ["Менеджер", "Администратор"]:
            self.add_status_combobox()

        # Клик по карточке для просмотра деталей
        self.orderFrame.mousePressEvent = lambda event: self.show_order_details()

    def add_status_combobox(self):
        layout = self.deliveryFrame.layout()

        self.status_combo = QComboBox(self)
        self.status_combo.setStyleSheet("border: 1px solid black;")

        statuses = get_order_statuses()
        for status_id, status_name in statuses:
            self.status_combo.addItem(status_name, status_id)
            if status_id == self.current_status_id:
                self.status_combo.setCurrentText(status_name)

        self.status_combo.currentIndexChanged.connect(self.on_status_changed)
        layout.addWidget(self.status_combo)

    # Обработчик изменения статуса
    def on_status_changed(self, index):
        new_status_id = self.status_combo.currentData()
        new_status_name = self.status_combo.currentText()

        # Если статус не изменился - ничего не делаем
        if new_status_id == self.current_status_id:
            return

        # Подтверждение изменения
        reply = QMessageBox.question(
            self, "Подтверждение",
            f"Изменить статус заказа №{self.order_id} на '{new_status_name}'?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )

        if reply == QMessageBox.StandardButton.Yes:
            try:
                update_order_status(self.order_id, new_status_id)
                self.current_status_id = new_status_id
                self.status_orderLabel.setText(f"Статус заказа: {new_status_name}")
                self.order_updated.emit()
                QMessageBox.information(self, "Успех", "Статус заказа обновлён!")
            except Exception as e:
                QMessageBox.critical(self, "Ошибка", f"Не удалось обновить статус: {str(e)}")
                # Возвращаем старый статус в ComboBox
                self.status_combo.blockSignals(True)
                for i in range(self.status_combo.count()):
                    if self.status_combo.itemData(i) == self.current_status_id:
                        self.status_combo.setCurrentIndex(i)
                        break
                self.status_combo.blockSignals(False)
        else:
            # Пользователь отказался - возвращаем старый статус в ComboBox
            self.status_combo.blockSignals(True)
            for i in range(self.status_combo.count()):
                if self.status_combo.itemData(i) == self.current_status_id:
                    self.status_combo.setCurrentIndex(i)
                    break
            self.status_combo.blockSignals(False)

    def show_order_details(self):
        items = get_order_items(self.order_id)

        if not items:
            QMessageBox.information(self, "Заказ", f"Заказ №{self.order_id} не содержит товаров.")
            return

        details = f"Заказ №{self.order_id}\n\n"
        details += "Товары в заказе:\n"

        total_sum = 0
        for item in items:
            item_id, product_id, article, product_name, quantity, price, discount, total_price = item
            details += f"Артикул: {article}\n"
            details += f"Товар: {product_name}\n"
            details += f"Кол-во: {quantity} шт.\n"
            details += f"Цена: {price} руб.\n"
            if discount > 0:
                details += f"Скидка: {discount}%\n"
                details += f"Итого: {int(total_price)} руб.\n"
            else:
                details += f"Итого: {price * quantity} руб.\n"
            total_sum += total_price if discount > 0 else price * quantity

        details += f"\nОбщая сумма заказа: {int(total_sum)} руб."
        QMessageBox.information(self, f"Детали заказа №{self.order_id}", details)

# 1) Принятие товара (кортежа) из БД и подготовка данных для отображения в интерфейсе
def prepare_product_data(product):
    # распаковка кортежа product в отдельные переменные
    (product_id, category, article, name, opisanie, proizvoditel, postavshik,
     price, izmerenie, kolvo, skidka, photo) = product

    # расчет цены со скидкой
    if skidka > 0: # если есть скидка, то считаем новую цену
        new_price = price - ((price * skidka) / 100)
    else:
        new_price = None

    # определение цвета карточки товара
    if kolvo == 0: # нет на складе
        bg_color = "blue"
    elif skidka > 15: # скидка > 15%
        bg_color = "#2E8B57"
    else: # обычный товар
        bg_color = None

    # словарь с готовыми данными
    return {
        "product_id": product_id,
        "category": category,
        "article": article,
        "name": name,
        "opisanie": opisanie,
        "proizvoditel": proizvoditel,
        "postavshik": postavshik,
        "price": price,
        "new_price": new_price,
        "izmerenie": izmerenie,
        "kolvo": kolvo,
        "skidka": skidka,
        "photo": photo,
        "bg_color": bg_color
    }

    # 11) Удаление товара
    def delete_product(self, product_id):
        from db import delete_product, product_in_orders, get_product_photo
        import os

        # Проверка: есть ли товар в заказах
        if product_in_orders(product_id):
            QMessageBox.warning(
                self,
                "Запрещено удаление",
                "Данный товар используется в заказах.\nСначала удалите его из заказов."
            )
            return

        # Подтверждение удаления
        reply = QMessageBox.question(self, "Подтверждение", "Вы уверены, что хотите удалить товар?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No)

        if reply != QMessageBox.StandardButton.Yes:
            return

        # Получаем имя файла картинки
        photo_name = get_product_photo(product_id)

        # Если фото есть и это НЕ заглушка
        if photo_name and photo_name != "picture.png":
            image_path = os.path.join("images", photo_name)

            if os.path.exists(image_path):
                os.remove(image_path)

        try:
            # Удаляем товар из БД
            delete_product(product_id)
            # Обновляем список товаров
            self.load_products()
        except Exception as e:
            QMessageBox.critical(
                self,
                "Ошибка удаления",
                f"Не удалось удалить товар.\nОшибка: {str(e)}"
            )
